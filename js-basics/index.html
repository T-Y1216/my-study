1. 变量的定义

练习 var、let、const 的声明方式与作用域差异。

// 变量的定义
var a = 10;
var a = 20;          // var 允许重复声明
console.log("var a:", a);

let b = 30;
b = 40;              // let 允许重新赋值，但不允许重复声明
console.log("let b:", b);

const c = 50;        // const 声明常量，必须初始化
console.log("const c:", c);

if (true) {
    var x = "var 在块级外可访问";
    let y = "let 只在块级内可访问";
    const z = "const 只在块级内可访问";
    console.log(y);
    console.log(z);
}
console.log("if 块外访问 var x:", x);

3. 表达式

练习算术表达式、比较表达式和逻辑表达式。

// 表达式
// 算术表达式
var sum = 10 + 5;
var diff = 10 - 5;
var product = 10 * 5;
var quotient = 10 / 3;
var remainder = 10 % 3;
var power = 2 ** 4;

console.log("算术:", sum, diff, product, quotient, remainder, power);

// 比较表达式
console.log(5 == "5");   // true，值相等
console.log(5 === "5"); // false，类型不同
console.log(10 > 3);
console.log(10 <= 3);

// 逻辑表达式
console.log(true && false);
console.log(true || false);
console.log(!true);

4. 流程控制

练习 if/else 条件判断、for 循环与 while 循环。

// 流程控制
// if/else 条件判断
var score = 85;
var grade;

if (score >= 90) {
    grade = "A（优秀）";
} else if (score >= 80) {
    grade = "B（良好）";
} else if (score >= 60) {
    grade = "C（及格）";
} else {
    grade = "D（不及格）";
}
console.log("成绩等级:", grade);

// for 循环：1到5的阶乘
var factorial = 1;
for (var i = 1; i <= 5; i++) {
    factorial *= i;
}
console.log("for 循环阶乘(1~5):", factorial);

// while 循环：累加到不小于10
var sum = 0;
var n = 1;
while (sum < 10) {
    sum += n;
    n++;
}
console.log("while 循环累加结果:", sum);
▶ 运行结果

5. 函数

练习函数声明、函数表达式、箭头函数以及回调函数。

// 函数
// 函数声明
function add(a, b) {
    return a + b;
}
console.log("函数声明 add(3,4):", add(3, 4));

// 函数表达式
var multiply = function(a, b) {
    return a * b;
};
console.log("函数表达式 multiply(5,6):", multiply(5, 6));

// 箭头函数
var divide = (a, b) => a / b;
console.log("箭头函数 divide(20,4):", divide(20, 4));

// 回调函数
function operate(a, b, callback) {
    return callback(a, b);
}
var result = operate(10, 3, function(x, y) {
    return x - y;
});
console.log("回调函数 operate(10,3,减法):", result);
